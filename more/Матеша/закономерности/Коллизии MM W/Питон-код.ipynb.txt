import csv, os
from itertools import combinations_with_replacement

N_MAX = 16
CSV_PATH = "/content/KHTSN_items.csv"
MD_PATH  = "/content/KHTSN_cores.md"
BUFFER_LIMIT = 10000

# ---------- Этап 1: CSV со всеми наборами без нулей ----------
print("Этап 1: генерация наборов...")
with open(CSV_PATH, "w", newline="", encoding="utf-8") as f:
    w = csv.writer(f)
    w.writerow(["W", "n", "S", "digits"])
    cnt = 0
    for k in range(1, N_MAX + 1):
        for digits in combinations_with_replacement(range(1, 10), k):
            S = sum(digits)
            Q = sum(d * d for d in digits)
            W = (S * S + Q) // 2
            w.writerow([W, k, S, ",".join(map(str, digits))])
            cnt += 1
print(f"  записано наборов: {cnt}")
print(f"  размер CSV: {os.path.getsize(CSV_PATH) / 1e6:.1f} МБ")

# ---------- Этап 2: чтение и сортировка ----------
print("Этап 2: чтение и сортировка...")
rows = []
with open(CSV_PATH, encoding="utf-8") as f:
    r = csv.reader(f)
    next(r)  # заголовок
    for W, n, S, dg in r:
        rows.append((int(W), int(n), int(S), dg))
rows.sort()
print(f"  отсортировано: {len(rows)} записей")

# ---------- Этап 3: группировка по W и потоковая запись ----------
print("Этап 3: группировка и запись отчёта...")

n_cores = n_equal = n_diff = 0
n_W_with_cores = 0
buffer = []

def flush():
    global buffer
    if buffer:
        with open(MD_PATH, "a", encoding="utf-8") as f:
            f.write("\n".join(buffer) + "\n")
        buffer.clear()

with open(MD_PATH, "w", encoding="utf-8") as f:
    f.write("# КХЦН: ядерные коллизии типа II\n\n")
    f.write("Величина W определена в файле [Величина W.md](https://github.com/Bizonrelax/Sigmawillow/blob/bbf9f477768bdc2344d3f8697cd45186d758cbcb/more/%D0%9C%D0%B0%D1%82%D0%B5%D1%88%D0%B0/%D0%BE%D0%BF%D1%80%D0%B5%D0%B4%D0%B5%D0%BB%D0%B5%D0%BD%D0%B8%D1%8F/%D0%92%D0%B5%D0%BB%D0%B8%D1%87%D0%B8%D0%BD%D0%B0%20W.md) как W = (S² + Q) / 2.\n\n")
    f.write("**Ядерная коллизия** — пара наборов цифр без нулей (все цифры от 1 до 9), у которых W совпадает, а суммы S различны. Любая коллизия типа II размера n получается из ядерной добавлением нулей.\n\n")
    f.write(f"Перебор: все наборы без нулей размера от 1 до {N_MAX}.\n\n")
    f.write("## Полная таблица\n\n")
    f.write("| W | Тип | min n | Набор A | S(A) | Набор B | S(B) |\n")
    f.write("|--:|:----|--:|:--------|-----:|:--------|-----:|\n")

i = 0
total = len(rows)
while i < total:
    j = i
    while j < total and rows[j][0] == rows[i][0]:
        j += 1
    W = rows[i][0]
    group = rows[i:j]
    had_pair = False
    for a in range(len(group)):
        for b in range(a + 1, len(group)):
            _, n1, S1, d1 = group[a]
            _, n2, S2, d2 = group[b]
            if S1 == S2:
                continue
            min_n = max(n1, n2)
            typ = "равные" if n1 == n2 else "разные"
            buffer.append(
                f"| {W} | {typ} | {min_n} | {{{d1}}} | {S1} | {{{d2}}} | {S2} |"
            )
            n_cores += 1
            if n1 == n2:
                n_equal += 1
            else:
                n_diff += 1
            had_pair = True
            if len(buffer) >= BUFFER_LIMIT:
                flush()
    if had_pair:
        n_W_with_cores += 1
    i = j
    if (i * 100) // total % 10 == 0:
        print(f"  прогресс: {i * 100 // total}%")

flush()

# ---------- Итоги ----------
with open(MD_PATH, "a", encoding="utf-8") as f:
    f.write("\n## Итоги\n\n")
    f.write("| Показатель | Значение |\n")
    f.write("|:-----------|---------:|\n")
    f.write(f"| Всего наборов без нулей (n ≤ {N_MAX}) | {len(rows)} |\n")
    f.write(f"| Всего ядерных пар | {n_cores} |\n")
    f.write(f"| С равными размерами сторон | {n_equal} |\n")
    f.write(f"| С разными размерами сторон | {n_diff} |\n")
    f.write(f"| Значений W с хотя бы одной коллизией | {n_W_with_cores} |\n")

print(f"\nГотово.")
print(f"  ядерных пар: {n_cores} (равных {n_equal}, разных {n_diff})")
print(f"  значений W с коллизией: {n_W_with_cores}")
print(f"  файл: {MD_PATH} ({os.path.getsize(MD_PATH) / 1e6:.1f} МБ)")

from google.colab import files
files.download(MD_PATH)
