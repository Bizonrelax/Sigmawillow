from collections import defaultdict
from itertools import combinations_with_replacement

def analyze_primitive_II(n_max):
    result = []
    for n in range(2, n_max + 1):
        groups = defaultdict(list)
        for digits in combinations_with_replacement(range(10), n):
            if all(d == 0 for d in digits):
                continue
            S = sum(digits)
            Q = sum(d * d for d in digits)
            W = (S * S + Q) // 2
            groups[W].append((S, digits))

        for W, items in groups.items():
            S_set = {s for s, _ in items}
            if len(items) < 2 or len(S_set) < 2:
                continue
            # чистая коллизия типа II: все S различны
            if len(S_set) != len(items):
                continue
            for i in range(len(items)):
                for j in range(i + 1, len(items)):
                    S1, d1 = items[i]
                    S2, d2 = items[j]
                    # проверка примитивности: нет общего нуля
                    if d1.count(0) > 0 and d2.count(0) > 0:
                        d1r = tuple(x for x in d1 if x != 0)
                        d2r = tuple(x for x in d2 if x != 0)
                        if len(d1r) == len(d2r):
                            Sr1 = sum(d1r); Sr2 = sum(d2r)
                            Qr1 = sum(x*x for x in d1r)
                            Qr2 = sum(x*x for x in d2r)
                            if (Sr1*Sr1 + Qr1) == (Sr2*Sr2 + Qr2) and Sr1 != Sr2:
                                continue  # не примитивная
                    result.append((n, W, d1, S1, d2, S2))
    return result

prims = analyze_primitive_II(16)
# Оставляем только те, что впервые появляются на своём n
seen = set()
lines = []
lines.append("# Полный список примитивных коллизий типа II")
lines.append("")
lines.append("| n первого появления | W | Набор A | S(A) | Набор B | S(B) |")
lines.append("|--:|--:|:--------|-----:|:--------|-----:|")
for n, W, d1, S1, d2, S2 in prims:
    key = (W, tuple(sorted(d1)), tuple(sorted(d2)))
    if key in seen:
        continue
    seen.add(key)
    lines.append(f"| {n} | {W} | {{{','.join(map(str,d1))}}} | {S1} | "
                 f"{{{','.join(map(str,d2))}}} | {S2} |")

report = "\n".join(lines)
with open("/content/primitive_II.md", "w", encoding="utf-8") as f:
    f.write(report)
print(f"Примитивных коллизий: {len(seen)}")
